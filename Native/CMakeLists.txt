cmake_minimum_required(VERSION 3.25)
project(LibtorrentNative LANGUAGES C CXX)

if(NOT DEFINED LIBTORRENT_SOURCE_DIR)
  message(FATAL_ERROR "LIBTORRENT_SOURCE_DIR is required")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(build_tools OFF CACHE BOOL "" FORCE)
set(python-bindings OFF CACHE BOOL "" FORCE)
set(logging OFF CACHE BOOL "" FORCE)
set(deprecated-functions ON CACHE BOOL "" FORCE)
set(streaming ON CACHE BOOL "" FORCE)
set(dht ON CACHE BOOL "" FORCE)
set(encryption ON CACHE BOOL "" FORCE)
set(extensions ON CACHE BOOL "" FORCE)
set(i2p OFF CACHE BOOL "" FORCE)
set(webtorrent OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)

add_subdirectory("${LIBTORRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/libtorrent" EXCLUDE_FROM_ALL)
target_compile_definitions(torrent-rasterbar PUBLIC TORRENT_HAVE_MMAP=0)
if(APPLE)
  foreach(property COMPILE_DEFINITIONS INTERFACE_COMPILE_DEFINITIONS)
    get_target_property(definitions torrent-rasterbar ${property})
    if(definitions)
      list(REMOVE_ITEM definitions TORRENT_USE_LIBCRYPTO)
      set_property(TARGET torrent-rasterbar PROPERTY ${property} "${definitions}")
    endif()
  endforeach()
  target_compile_definitions(torrent-rasterbar PUBLIC TORRENT_USE_COMMONCRYPTO=1)
endif()

add_library(LibtorrentNative STATIC src/LibtorrentNative.cpp)
target_include_directories(LibtorrentNative PUBLIC include)
target_link_libraries(LibtorrentNative PRIVATE torrent-rasterbar)
target_compile_options(LibtorrentNative PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
set_target_properties(LibtorrentNative PROPERTIES OUTPUT_NAME LibtorrentNative)
